cmake_minimum_required(VERSION 3.16)
project(robot_motion_safety)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp geometry_msgs std_msgs std_srvs ros_gz_interfaces)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(cmd_vel_arbiter src/cmd_vel_arbiter.cpp)
target_compile_features(cmd_vel_arbiter PRIVATE cxx_std_17)
target_include_directories(cmd_vel_arbiter PRIVATE include)
target_compile_options(cmd_vel_arbiter PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cmd_vel_arbiter ${dependencies})
install(TARGETS cmd_vel_arbiter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(safety_test test/safety_test.cpp)
  target_include_directories(safety_test PRIVATE include)
  target_compile_features(safety_test PRIVATE cxx_std_17)
  ament_target_dependencies(safety_test geometry_msgs)
endif()
ament_package()
